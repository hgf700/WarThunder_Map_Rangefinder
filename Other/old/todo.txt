ThreadPoolExecutor — do obsługi zadań “cięższych”, ale nie stricte CPU (np. YOLO, odczyt plików, backend),
Queue — do komunikacji między backendem a UI (np. przesył wyników, błędów),

przyklad
from concurrent.futures import ThreadPoolExecutor, as_completed
from queue import Queue
import threading
import time

# Kolejka wyników do UI
result_queue = Queue()

def yolo_task(number):
    """Symulacja detekcji YOLO"""
    time.sleep(1)
    if number == 2:
        raise ValueError("YOLO padło na klatce 2")
    return f"Detekcja zakończona: {number}"

def manage_results():
    """Wątek UI / InGameUI monitorujący wyniki"""
    while True:
        result = result_queue.get()
        if result == "STOP":
            print("[UI] Zatrzymano monitorowanie wyników.")
            break
        print(f"[UI] Nowy wynik: {result}")

def main():
    ui_thread = threading.Thread(target=manage_results, daemon=True)
    ui_thread.start()

    with ThreadPoolExecutor(max_workers=3) as executor:
        futures = [executor.submit(yolo_task, i) for i in range(5)]

        for future in as_completed(futures):
            try:
                result = future.result()
                result_queue.put(result)
            except Exception as e:
                result_queue.put(f"Błąd: {e}")

    result_queue.put("STOP")

main()
