#OCR
from ultralytics import YOLO
import pytesseract, cv2
model = YOLO('runs/detect/train_new/weights/best.pt')
results = model('map_071.png', imgsz=800)
for r in results:
    for box, cls in zip(r.boxes.xyxy, r.boxes.cls):
        if int(cls) == 2:
            x1, y1, x2, y2 = map(int, box)
            crop = r.orig_img[y1:y2, x1:x2]
            value = pytesseract.image_to_string(crop, config='--psm 8 digits')
            print(f"Metry: {value}m")
